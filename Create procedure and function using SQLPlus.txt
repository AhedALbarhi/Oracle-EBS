- create procedure and function in the database:
- function:
    
SQL>  CREATE OR REPLACE FUNCTION AHED_FN_TEST RETURN VARCHAR2 IS
       V_MSG VARCHAR2(2000);
       BEGIN
         V_MSG:= 'WELCOME TO ORACLE';
         RETURN V_MSG;
       END;
       /
------
-call the function:

SQL> SELECT AHED_FN_TEST FROM DUAL;

     END;
