BEGIN

EXCEPTION
WHEN OTHERS THEN
FND_FILE.PUT_LINE(FND_FILE.OUTPUT, ' CHECK YOUR PROGRAM: '
||ERRBUF|| ' ERRCODE = '||RETCODE);
END;
